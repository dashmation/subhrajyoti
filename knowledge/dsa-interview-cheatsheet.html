Javapublic int[] twoSum(int[] nums, int target) {
    Map<Integer, Integer> map = new HashMap<>();
    for (int i = 0; i < nums.length; i++) {
        int comp = target - nums[i];
        if (map.containsKey(comp)) return new int[]{map.get(comp), i};
        map.put(nums[i], i);
    }
    return new int[]{};
}

        seen[num] = i
    return []

When to use: Use on sorted arrays or when you need to compare elements from both ends or track two positions. Signals: "sorted array", "pair with sum", "remove duplicates in-place", "container with most water", "move zeros".

Javapublic int[] twoSumSorted(int[] nums, int target) {
    int l = 0, r = nums.length - 1;
    while (l < r) {
        int sum = nums[l] + nums[r];
        if (sum == target) return new int[]{l + 1, r + 1};
        else if (sum < target) l++;
        else r--;
    }
    return new int[]{};
}
Pythondef two_sum_sorted(nums, target):

        if s == target: return [l + 1, r + 1]
        elif s < target: l += 1
        else: r -= 1
    return []

6. Sliding Window

When to use: When you need to find a contiguous subarray/substring satisfying some condition. Fixed-size window for exact length; variable-size for "minimum/maximum/longest" conditions. Signals: "consecutive elements", "substring of length k", "longest without repeating", "minimum window".

Pattern / TechniqueTime / SpaceExample
Fixed Window: Max Sum Subarray of Size KO(n) / O(1)
▶ Show
Javapublic int maxSumSubarray(int[] nums, int k) {
    int windowSum = 0, maxSum = 0;
    for (int i = 0; i < nums.length; i++) {
        windowSum += nums[i];
        if (i >= k) windowSum -= nums[i - k];
        if (i >= k - 1) maxSum = Math.max(maxSum, windowSum);
    }
    return maxSum;
}
Pythondef max_sum_subarray(nums, k):
    window_sum = sum(nums[:k])
    max_sum = window_sum
    for i in range(k, len(nums)):
        window_sum += nums[i] - nums[i - k]
        max_sum = max(max_sum, window_sum)
    return max_sum

Javapublic int lengthOfLongestSubstring(String s) {

        char c = s.charAt(right);
        if (map.containsKey(c)) left = Math.max(left, map.get(c) + 1);
        map.put(c, right);
        maxLen = Math.max(maxLen, right - left + 1);

Javapublic String minWindow(String s, String t) {
    int[] need = new int[128], have = new int[128];
    for (char c : t.toCharArray()) need[c]++;
    int required = 0;
    for (int n : need) if (n > 0) required++;
    int formed = 0, l = 0, minLen = Integer.MAX_VALUE, start = 0;
    for (int r = 0; r < s.length(); r++) {
        char c = s.charAt(r);
        have[c]++;
        if (need[c] > 0 && have[c] == need[c]) formed++;
        while (formed == required) {
            if (r - l + 1 < minLen) { minLen = r - l + 1; start = l; }
            char lc = s.charAt(l);
            have[lc]--;
            if (need[lc] > 0 && have[lc] < need[lc]) formed--;
            l++;
        }
    }
    return minLen == Integer.MAX_VALUE ? "" : s.substring(start, start + minLen);
}
Pythondef min_window(s, t):
    from collections import Counter
    need = Counter(t)
    missing = len(t)
    l = start = 0
    min_len = float('inf')

7. Prefix Sum

When to use: When you need O(1) range sum queries after preprocessing, or need to count subarrays with a given sum. Signals: "sum between indices", "subarray sum equals k", "number of subarrays with sum", "range query". Often combined with a hashmap for counting.
