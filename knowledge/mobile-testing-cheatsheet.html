import XCTest
@testable import MyApp

class UserServiceTests: XCTestCase {
    var sut: UserService!

    override func setUp() {
        super.setUp()
        sut = UserService(api: MockAPIClient())
    }

    override func tearDown() {
        sut = nil
        super.tearDown()
    }

    func testFetchUser_ReturnsValidUser() async throws {
        let user = try await sut.fetchUser(id: "123")
        XCTAssertEqual(user.name, "John Doe")
        XCTAssertEqual(user.email, "john@test.com")
    }

    func testFetchUser_InvalidID_ThrowsError() async {
        do {
            _ = try await sut.fetchUser(id: "invalid")
            XCTFail("Expected error not thrown")
        } catch {
            XCTAssertEqual(error as? AppError, .userNotFound)
        }
    }
}

Page Object ModelDesign pattern for maintainable test code. Separate page structure from test logic.
▶ Show
// Page Object - LoginPage.java
public class LoginPage {
    private AndroidDriver driver;

    @AndroidFindBy(accessibility = "email_field")
    private WebElement emailField;

    @AndroidFindBy(accessibility = "password_field")
    private WebElement passwordField;

    @AndroidFindBy(accessibility = "login_button")
    private WebElement loginButton;

    public LoginPage(AndroidDriver driver) {
        this.driver = driver;
        PageFactory.initElements(
            new AppiumFieldDecorator(driver), this);
    }

    public HomePage login(String email, String password) {
        emailField.sendKeys(email);
        passwordField.sendKeys(password);
        loginButton.click();
        return new HomePage(driver);
    }
}

// Test using Page Object
@Test
public void testLogin() {

    HomePage home = loginPage.login("user@test.com", "pass");
    assertTrue(home.isWelcomeDisplayed());

Unit Tests (flutter_test)Test individual Dart functions and classes. Fast, no UI rendering needed.

// test/user_service_test.dart
import 'package:flutter_test/flutter_test.dart';
import 'package:mockito/mockito.dart';
import 'package:my_app/services/user_service.dart';

class MockApi extends Mock implements ApiClient {}

void main() {
  late UserService service;
  late MockApi mockApi;

  setUp(() {
    mockApi = MockApi();
    service = UserService(api: mockApi);
  });

  group('UserService', () {
    test('fetchUser returns user on success', () async {
      when(mockApi.getUser('123'))
          .thenAnswer((_) async => User(name: 'John'));

      final user = await service.fetchUser('123');
      expect(user.name, equals('John'));
      verify(mockApi.getUser('123')).called(1);
    });

    test('fetchUser throws on network error', () {
      when(mockApi.getUser(any))
          .thenThrow(NetworkException());

      expect(
        () => service.fetchUser('123'),
        throwsA(isA<NetworkException>()),
      );
    });
  });
}

▶ Show

// By index (1-based)
driver.findElement(AppiumBy.iOSClassChain(
    "**/XCUIElementTypeCell[3]"));

// Nested hierarchy
driver.findElement(AppiumBy.iOSClassChain(

// Last element
driver.findElement(AppiumBy.iOSClassChain(
    "**/XCUIElementTypeButton[-1]"));

// Multiple conditions
driver.findElement(AppiumBy.iOSClassChain(
    "**/XCUIElementTypeButton[`label CONTAINS 'Save' AND visible == 1`]"));

// Descendant search (any depth)
driver.findElement(AppiumBy.iOSClassChain(
    "**/XCUIElementTypeScrollView/**/XCUIElementTypeButton[`label == 'Add'`]"));

// Android - by resource-id

// iOS XPath
driver.findElement(By.xpath(

// Relative XPath (sibling, parent)
driver.findElement(By.xpath(

    "/following-sibling::android.widget.EditText"));

    "//*[contains(@text, 'Welcome')]"));

1. Accessibility ID
2. ID / Resource ID
3. iOS Predicate / Class Chain
4. Android UIAutomator
5. Class Name
6. XPath (SLOWEST - avoid in loops)

// Custom URLProtocol for intercepting requests
class NetworkLogger: URLProtocol {
    override class func canInit(with request: URLRequest) -> Bool {
        // Intercept all HTTP/HTTPS requests
        guard let scheme = request.url?.scheme else { return false }
        return ["http", "https"].contains(scheme)
    }

    override class func canonicalRequest(
        for request: URLRequest
    ) -> URLRequest {
        return request
    }

    override func startLoading() {
        // Log the request
        print("📡 \(request.httpMethod ?? "") \(request.url?.absoluteString ?? "")")
        if let body = request.httpBody {
            print("   Body: \(String(data: body, encoding: .utf8) ?? "")")
        }

        // Forward to actual network
        let session = URLSession(configuration: .default,
                                  delegate: self as? URLSessionDelegate,
                                  delegateQueue: nil)
        let task = session.dataTask(with: request) { data, response, error in
            if let response = response {
                self.client?.urlProtocol(self, didReceive: response,
                    cacheStoragePolicy: .notAllowed)
            }
            if let data = data {
                self.client?.urlProtocol(self, didLoad: data)
            }
            self.client?.urlProtocolDidFinishLoading(self)
        }
        task.resume()
    }

    override func stopLoading() {}
}

// Register in AppDelegate / test setUp:
URLProtocol.registerClass(NetworkLogger.self)

let session = URLSession(configuration: config)

What is SSL/Certificate Pinning?
App only trusts specific certificates/public keys,
rejecting valid but unexpected certificates (like proxy certs).

Testing pinning is working:
1. Setup Charles Proxy / Proxyman
2. Enable SSL proxying for your domain
3. Expected result: App should FAIL to connect
   (proves pinning rejects the proxy certificate)
4. If requests go through → pinning is broken!

Bypass pinning for testing (debug builds only):

Method 1: Build configuration
#if DEBUG
// Disable pinning in debug builds
let session = URLSession(configuration: .default,
    delegate: TrustAllCertsDelegate(), delegateQueue: nil)
#endif

Method 2: Using Frida (dynamic instrumentation)
frida -U -n MyApp -l ssl-pinning-bypass.js

Method 3: Using objection (Frida wrapper)
objection -g MyApp explore
... ios sslpinning disable

Common pinning implementations:
- TrustKit: open-source library for iOS
- Alamofire ServerTrustManager
- URLSessionDelegate: challenge handler
- NSAppTransportSecurity with pinned domains

Interview: "How do you test API calls if app has SSL pinning?"
→ Use debug build flag to disable pinning in test environments.
  Or use Frida/objection for runtime bypass on test builds.
  NEVER ship with pinning disabled.

// Basic expectation pattern
func testAsyncNetworkCall() {
    let expectation = expectation(description: "API call completes")

    apiService.fetchUser(id: "123") { result in
        switch result {
        case .success(let user):
            XCTAssertEqual(user.name, "John")
        case .failure(let error):
            XCTFail("Unexpected error: \(error)")
        }
        expectation.fulfill()
    }

    waitForExpectations(timeout: 5.0)
}

// Swift async/await (cleaner, iOS 13+)
func testAsyncAwaitCall() async throws {
    let user = try await apiService.fetchUser(id: "123")
    XCTAssertEqual(user.name, "John")
}

// Multiple expectations
func testParallelRequests() {
    let exp1 = expectation(description: "Users loaded")
    let exp2 = expectation(description: "Posts loaded")

    loadUsers { exp1.fulfill() }
    loadPosts { exp2.fulfill() }

    wait(for: [exp1, exp2], timeout: 10.0, enforceOrder: false)
}

// Inverted expectation (something should NOT happen)
func testNoUnexpectedCallback() {
    let noCallback = expectation(description: "Should not fire")
    noCallback.isInverted = true

    cancelledOperation.onComplete = { noCallback.fulfill() }
    cancelledOperation.cancel()

    waitForExpectations(timeout: 2.0) // Passes if NOT fulfilled
}

━━━ TOP iOS TESTING INTERVIEW Q&A ━━━

Q: How to capture network logs on iOS?
A: Charles Proxy / Proxyman (HTTPS with cert install),
   CFNetwork Diagnostics env var (quick, no setup),
   URLProtocol subclass (in-app), mitmproxy (CI/CD)

Q: How to capture system logs?
A: Console.app (GUI), xcrun simctl spawn log stream (CLI),
   idevicesyslog (real device, no Xcode), os_log API (in code)

Q: How to find memory leaks?
A: Instruments → Leaks/Allocations, Xcode Memory Graph (Cmd+Shift+D),
   Look for retain cycles: [weak self] in closures, weak delegates

Q: How to debug crashes?
A: Xcode Organizer (prod), dSYM + atos (symbolicate),
   Enable Zombies (EXC_BAD_ACCESS), Address Sanitizer

Q: How to measure app performance?
A: Instruments (Time Profiler, Core Animation), MetricKit (prod),
   XCTest measure {} with baselines in CI

Q: How to test on slow network?
A: Network Link Conditioner (device/Mac), Charles throttling,
   Reachability API for offline mode

Q: Simulator vs real device?
A: Simulator for UI/logic. Real device for performance, memory,
   hardware features, battery. ALWAYS profile on real device.

Q: What's a dSYM file?
A: Debug symbols mapping addresses to function names/lines.
   Required for crash report symbolication. Upload to App Store.

Q: What causes 0x8badf00d?
A: Watchdog killed app — main thread blocked too long during
   launch, backgrounding, or system events.

Q: How to bypass SSL pinning for testing?
A: Debug build flag, Frida/objection runtime bypass, or
   configure test environment with pinning disabled.
