7. Waits BOTH

Waits handle timing issues — the #1 cause of flaky tests. Playwright's auto-wait eliminates most explicit waits.

ConceptSelenium (Java)Playwright (TypeScript)Example
Auto-waitNot built-inBuilt-in (all actions auto-wait)
▶ Show
// Must add explicit waits manually
// click() auto-waits for visible + enabled
await page.locator('#btn').click(); // waits automatically
Implicit waitdriver.manage().timeouts().implicitlyWait(...)Not needed
▶ Show
driver.manage().timeouts().implicitlyWait(Duration.ofSeconds(10));
// Not needed — Playwright auto-waits on all actions
Explicit waitnew WebDriverWait(driver, ...).until(...)await locator.waitFor()
▶ Show
WebDriverWait wait = new WebDriverWait(driver, Duration.ofSeconds(10));
wait.until(ExpectedConditions.visibilityOfElementLocated(By.id("msg")));
await page.locator('#msg').waitFor({ state: 'visible' });
Wait for textEC.textToBePresentInElementexpect(locator).toHaveText(...)
▶ Show
wait.until(ExpectedConditions.textToBePresentInElementLocated(
    By.id("status"), "Done"));
await expect(page.locator('#status')).toHaveText('Done');
Wait hiddenEC.invisibilityOfElementwaitFor({ state: 'hidden' })
▶ Show
wait.until(ExpectedConditions.invisibilityOfElementLocated(By.id("loader")));
await page.locator('#loader').waitFor({ state: 'hidden' });
Wait for URLEC.urlContains("...")await page.waitForURL('...')
▶ Show
wait.until(ExpectedConditions.urlContains("/dashboard"));
await page.waitForURL('**/dashboard');
Custom timeoutPass Duration to wait{ timeout: ms } param
▶ Show
new WebDriverWait(driver, Duration.ofSeconds(30))
    .until(EC.presenceOfElementLocated(By.id("el")));
await page.locator('#el').click({ timeout: 30000 });

    private By username = By.id("user");
    private By password = By.id("pass");

        this.driver = driver;

import { Page, Locator } from '@playwright/test';

export class LoginPage {
  readonly page: Page;
  readonly username: Locator;
  readonly password: Locator;
  readonly loginBtn: Locator;

  constructor(page: Page) {
    this.page = page;
    this.username = page.locator('#user');
    this.password = page.locator('#pass');
    this.loginBtn = page.locator('#login');
  }
}
