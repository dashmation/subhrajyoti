1. Basics (SELECT, WHERE, ORDER BY, LIMIT)

SELECT is the foundation of SQL — retrieving and shaping data from tables is what you'll do 90% of the time.

Command / SyntaxDescriptionExample
SELECTRetrieve columns from a table
▶ Show
SELECT first_name, last_name
FROM employees;

SELECT * FROM employees;  -- all columns

SELECT DISTINCT department
FROM employees;  -- unique values only
AS (Alias)Rename columns or tables in output
▶ Show
SELECT first_name AS fname,
       salary * 12 AS annual_salary
FROM employees AS e;  -- table alias
WHEREFilter rows based on conditions
▶ Show
SELECT * FROM employees
WHERE department = 'Engineering';

SELECT * FROM employees
WHERE salary > 50000;

SELECT * FROM employees
WHERE hire_date >= '2023-01-01';
ORDER BYSort results (ASC default, DESC for reverse)
▶ Show
SELECT * FROM employees
ORDER BY salary DESC;  -- highest first

SELECT * FROM employees
ORDER BY department ASC, salary DESC;  -- multi-col

SELECT * FROM employees
ORDER BY 2;  -- by column position
LIMIT / OFFSETRestrict number of rows returned
▶ Show
SELECT * FROM employees
LIMIT 10;  -- first 10 rows

SELECT * FROM employees
LIMIT 10 OFFSET 20;  -- skip 20, get next 10

-- SQL Server syntax:
SELECT TOP 10 * FROM employees;

-- Oracle syntax:
SELECT * FROM employees
FETCH FIRST 10 ROWS ONLY;
CASE WHENConditional logic in queries
▶ Show
SELECT first_name,
  CASE
    WHEN salary > 100000 THEN 'High'
    WHEN salary > 50000  THEN 'Mid'
    ELSE 'Low'
  END AS salary_band
FROM employees;
COALESCE / NULLIFHandle NULL values
▶ Show
SELECT COALESCE(phone, email, 'N/A') AS contact
FROM employees;  -- first non-null value

SELECT NULLIF(bonus, 0) AS bonus
FROM employees;  -- returns NULL if bonus = 0

2. Filtering (AND/OR, IN, BETWEEN, LIKE, IS NULL)

Filtering narrows results to exactly what you need — essential for performance and relevance.

Command / SyntaxDescriptionExample
AND / ORCombine multiple conditions
▶ Show
SELECT * FROM employees
WHERE department = 'Sales'
  AND salary > 60000;

SELECT * FROM employees
WHERE department = 'Sales'
  OR department = 'Marketing';

-- Use parentheses for precedence:
SELECT * FROM employees
WHERE (dept = 'Sales' OR dept = 'HR')
  AND salary > 50000;
INMatch against a list of values
▶ Show
SELECT * FROM employees
WHERE department IN ('Sales', 'Marketing', 'HR');

-- NOT IN:
SELECT * FROM employees
WHERE department NOT IN ('IT', 'Finance');

-- IN with subquery:
SELECT * FROM employees
WHERE dept_id IN (
  SELECT id FROM departments WHERE active = 1
);
BETWEENFilter within a range (inclusive)
▶ Show
SELECT * FROM employees
WHERE salary BETWEEN 40000 AND 80000;

SELECT * FROM orders
WHERE order_date BETWEEN '2024-01-01' AND '2024-12-31';

-- NOT BETWEEN:
SELECT * FROM products
WHERE price NOT BETWEEN 10 AND 50;
LIKEPattern matching with wildcards
▶ Show
-- % = any number of characters
SELECT * FROM employees
WHERE name LIKE 'John%';  -- starts with John

SELECT * FROM employees
WHERE email LIKE '%@gmail.com';  -- ends with

-- _ = exactly one character
SELECT * FROM products
WHERE code LIKE 'A_B%';  -- A + any char + B...

-- Case-insensitive (PostgreSQL):
SELECT * FROM employees
WHERE name ILIKE '%smith%';
IS NULL / IS NOT NULLCheck for NULL values
▶ Show
SELECT * FROM employees
WHERE manager_id IS NULL;  -- no manager

SELECT * FROM employees
WHERE phone IS NOT NULL;

-- Common mistake (this does NOT work):
-- WHERE manager_id = NULL  ← always false!
EXISTSCheck if subquery returns any rows
▶ Show
SELECT * FROM customers c
WHERE EXISTS (
  SELECT 1 FROM orders o
  WHERE o.customer_id = c.id
);  -- customers who have placed orders

SELECT * FROM customers c
WHERE NOT EXISTS (
  SELECT 1 FROM orders o
  WHERE o.customer_id = c.id
);  -- customers with no orders

3. Aggregation (COUNT, SUM, AVG, MIN, MAX, GROUP BY, HAVING)

Aggregation summarizes data into meaningful statistics — counts, totals, averages across groups.

Command / SyntaxDescriptionExample
COUNTCount rows or non-null values
▶ Show
SELECT COUNT(*) FROM employees;  -- all rows

SELECT COUNT(phone) FROM employees;  -- non-null only

SELECT COUNT(DISTINCT department)
FROM employees;  -- unique departments
SUM / AVGSum or average of numeric column
▶ Show
SELECT SUM(salary) AS total_payroll
FROM employees;

SELECT AVG(salary) AS avg_salary
FROM employees
WHERE department = 'Engineering';

SELECT department,
       SUM(salary) AS dept_total,
       AVG(salary) AS dept_avg
FROM employees
GROUP BY department;
MIN / MAXMinimum or maximum value
▶ Show
SELECT MIN(salary), MAX(salary)
FROM employees;

SELECT MIN(hire_date) AS earliest,
       MAX(hire_date) AS latest
FROM employees;
GROUP BYGroup rows for aggregate calculations
▶ Show
SELECT department, COUNT(*) AS headcount
FROM employees
GROUP BY department;

-- Multiple grouping columns:
SELECT department, job_title, AVG(salary)
FROM employees
GROUP BY department, job_title;

-- With ORDER BY:
SELECT department, COUNT(*) AS cnt
FROM employees
GROUP BY department
ORDER BY cnt DESC;
HAVINGFilter groups (WHERE for aggregates)
▶ Show
SELECT department, COUNT(*) AS cnt
FROM employees
GROUP BY department
HAVING COUNT(*) > 5;  -- depts with more than 5

SELECT department, AVG(salary) AS avg_sal
FROM employees
GROUP BY department
HAVING AVG(salary) > 70000
ORDER BY avg_sal DESC;
GROUPING SETS / ROLLUP / CUBEMultiple grouping levels in one query
▶ Show
-- ROLLUP: hierarchical subtotals
SELECT department, job_title, SUM(salary)
FROM employees
GROUP BY ROLLUP(department, job_title);

-- CUBE: all combinations of subtotals
SELECT department, job_title, SUM(salary)
FROM employees
GROUP BY CUBE(department, job_title);

-- GROUPING SETS: specific combos
SELECT department, job_title, SUM(salary)
FROM employees
GROUP BY GROUPING SETS(
  (department, job_title),
  (department),
  ()
);

4. Joins (INNER, LEFT, RIGHT, FULL, CROSS, SELF)

Joins combine related data from multiple tables — the key to working with normalized relational databases.

Command / SyntaxDescriptionExample
INNER JOINRows that match in both tables
▶ Show
SELECT e.name, d.dept_name
FROM employees e
INNER JOIN departments d
  ON e.dept_id = d.id;

-- Multiple join conditions:
SELECT o.id, p.name, oi.quantity
FROM orders o
JOIN order_items oi ON o.id = oi.order_id
JOIN products p ON oi.product_id = p.id;
LEFT JOINAll rows from left + matching from right (NULL if no match)
▶ Show
SELECT e.name, d.dept_name
FROM employees e
LEFT JOIN departments d
  ON e.dept_id = d.id;
-- Shows all employees, even without dept

-- Find unmatched rows (anti-join):
SELECT c.name
FROM customers c
LEFT JOIN orders o ON c.id = o.customer_id
WHERE o.id IS NULL;  -- customers with no orders
RIGHT JOINAll rows from right + matching from left
▶ Show
SELECT e.name, d.dept_name
FROM employees e
RIGHT JOIN departments d
  ON e.dept_id = d.id;
-- Shows all departments, even with no employees

-- Equivalent to swapping tables in LEFT JOIN
FULL OUTER JOINAll rows from both tables (NULL where no match)
▶ Show
SELECT e.name, d.dept_name
FROM employees e
FULL OUTER JOIN departments d
  ON e.dept_id = d.id;
-- All employees AND all departments
-- NULLs on both sides where no match

-- Find orphans on both sides:
SELECT e.name, d.dept_name
FROM employees e
FULL OUTER JOIN departments d
  ON e.dept_id = d.id
WHERE e.id IS NULL OR d.id IS NULL;
CROSS JOINCartesian product (every combination)
▶ Show
SELECT s.size, c.color
FROM sizes s
CROSS JOIN colors c;
-- If sizes has 3 rows and colors has 4,
-- result has 12 rows (3 × 4)

-- Useful for generating date series:
SELECT d.date, p.product_id
FROM dates d
CROSS JOIN products p;
SELF JOINJoin a table to itself
▶ Show
-- Employee and their manager:
SELECT e.name AS employee,
       m.name AS manager
FROM employees e
LEFT JOIN employees m
  ON e.manager_id = m.id;

-- Find duplicates:
SELECT a.id, b.id, a.email
FROM users a
JOIN users b
  ON a.email = b.email
  AND a.id < b.id;
NATURAL JOINAuto-join on columns with same name
▶ Show
SELECT * FROM employees
NATURAL JOIN departments;
-- Joins on all columns with matching names
-- ⚠️ Fragile — avoid in production code

-- USING clause (explicit shared column):
SELECT * FROM employees
JOIN departments USING (dept_id);

5. Subqueries (Scalar, Column, Table, Correlated)

Subqueries embed one query inside another — useful when you need intermediate results for filtering or comparison.

Command / SyntaxDescriptionExample
Scalar SubqueryReturns a single value; usable in SELECT or WHERE
▶ Show
SELECT name, salary,
  (SELECT AVG(salary) FROM employees) AS avg_sal
FROM employees;

SELECT * FROM employees
WHERE salary > (
  SELECT AVG(salary) FROM employees
);
Column SubqueryReturns a list of values (use with IN, ANY, ALL)
▶ Show
SELECT * FROM employees
WHERE dept_id IN (
  SELECT id FROM departments
  WHERE location = 'London'
);

-- ANY: true if condition matches any value
SELECT * FROM products
WHERE price > ANY (
  SELECT price FROM products WHERE category = 'A'
);

-- ALL: true if condition matches all values
SELECT * FROM products
WHERE price > ALL (
  SELECT price FROM products WHERE category = 'B'
);
Table Subquery (Derived Table)Returns a result set used as a virtual table in FROM
▶ Show
SELECT dept, avg_sal
FROM (
  SELECT department AS dept,
         AVG(salary) AS avg_sal
  FROM employees
  GROUP BY department
) AS dept_stats
WHERE avg_sal > 60000;
Correlated SubqueryReferences outer query — runs once per outer row
▶ Show
-- Employees earning above their dept average:
SELECT e.name, e.salary, e.department
FROM employees e
WHERE e.salary > (
  SELECT AVG(e2.salary)
  FROM employees e2
  WHERE e2.department = e.department
);

-- Latest order per customer:
SELECT * FROM orders o
WHERE o.order_date = (
  SELECT MAX(o2.order_date)
  FROM orders o2
  WHERE o2.customer_id = o.customer_id
);

6. Data Manipulation (INSERT, UPDATE, DELETE, UPSERT)

DML commands modify data — inserting new records, updating existing ones, and removing obsolete data.

Command / SyntaxDescriptionExample
INSERT INTOAdd new rows to a table
▶ Show
-- Single row:
INSERT INTO employees (name, dept, salary)
VALUES ('Alice', 'Engineering', 85000);

-- Multiple rows:
INSERT INTO employees (name, dept, salary)
VALUES
  ('Bob', 'Sales', 60000),
  ('Carol', 'HR', 55000),
  ('Dave', 'Engineering', 90000);

-- Insert from SELECT:
INSERT INTO archive_employees
SELECT * FROM employees
WHERE terminated = true;
UPDATEModify existing rows
▶ Show
UPDATE employees
SET salary = 95000
WHERE id = 42;

-- Update multiple columns:
UPDATE employees
SET salary = salary * 1.10,
    title = 'Senior Engineer'
WHERE department = 'Engineering'
  AND years_exp >= 5;

-- Update with JOIN (PostgreSQL):
UPDATE employees e
SET dept_name = d.name
FROM departments d
WHERE e.dept_id = d.id;
DELETERemove rows from a table
▶ Show
DELETE FROM employees
WHERE id = 42;

DELETE FROM logs
WHERE created_at < '2023-01-01';

-- Delete with subquery:
DELETE FROM orders
WHERE customer_id IN (
  SELECT id FROM customers
  WHERE inactive = true
);

-- ⚠️ Delete ALL rows (careful!):
DELETE FROM temp_data;
UPSERTInsert or update if row exists
▶ Show
-- PostgreSQL (ON CONFLICT):
INSERT INTO products (sku, name, price)
VALUES ('ABC123', 'Widget', 9.99)
ON CONFLICT (sku)
DO UPDATE SET
  name = EXCLUDED.name,
  price = EXCLUDED.price;

-- MySQL (ON DUPLICATE KEY):
INSERT INTO products (sku, name, price)
VALUES ('ABC123', 'Widget', 9.99)
ON DUPLICATE KEY UPDATE
  name = VALUES(name),
  price = VALUES(price);

-- SQL Server (MERGE):
MERGE INTO products AS target
USING (VALUES ('ABC123','Widget',9.99))
  AS src(sku, name, price)
ON target.sku = src.sku
WHEN MATCHED THEN UPDATE SET
  name = src.name, price = src.price
WHEN NOT MATCHED THEN INSERT
  (sku, name, price) VALUES
  (src.sku, src.name, src.price);

7. Data Definition (CREATE, ALTER, DROP, TRUNCATE)

DDL commands define database structure — creating tables, altering schemas, managing the data model.

Command / SyntaxDescriptionExample
CREATE TABLEDefine a new table with columns and types
▶ Show
CREATE TABLE employees (
  id        SERIAL PRIMARY KEY,
  name      VARCHAR(100) NOT NULL,
  email     VARCHAR(255) UNIQUE,
  salary    DECIMAL(10,2) DEFAULT 0,
  dept_id   INT REFERENCES departments(id),
  hire_date DATE DEFAULT CURRENT_DATE,
  active    BOOLEAN DEFAULT true
);

-- Create from another table:
CREATE TABLE emp_backup AS
SELECT * FROM employees;
ALTER TABLEModify table structure
▶ Show
-- Add column:
ALTER TABLE employees
ADD COLUMN phone VARCHAR(20);

-- Drop column:
ALTER TABLE employees
DROP COLUMN phone;

-- Rename column:
ALTER TABLE employees
RENAME COLUMN name TO full_name;

-- Change type:
ALTER TABLE employees
ALTER COLUMN salary TYPE NUMERIC(12,2);

-- Rename table:
ALTER TABLE employees RENAME TO staff;
DROP TABLEDelete a table and all its data permanently
▶ Show
DROP TABLE employees;  -- error if not exists

DROP TABLE IF EXISTS employees;  -- safe

-- Drop with dependent objects:
DROP TABLE departments CASCADE;
-- removes FKs referencing this table
TRUNCATERemove all rows quickly (no logging per row)
▶ Show
TRUNCATE TABLE logs;

-- Reset auto-increment:
TRUNCATE TABLE logs RESTART IDENTITY;

-- Truncate with FK dependencies:
TRUNCATE TABLE orders CASCADE;

-- TRUNCATE vs DELETE:
-- TRUNCATE: faster, resets identity, can't WHERE
-- DELETE:   slower, row-level log, can filter
Common Data TypesFrequently used column types across databases
▶ Show
-- Numeric:
INT, BIGINT, SMALLINT, SERIAL
DECIMAL(p,s), NUMERIC(p,s), FLOAT, REAL

-- Text:
CHAR(n), VARCHAR(n), TEXT

-- Date/Time:
DATE, TIME, TIMESTAMP, INTERVAL

-- Boolean:
BOOLEAN  -- true/false/null

-- Binary:
BYTEA (Postgres), BLOB (MySQL/Oracle)

-- JSON:
JSON, JSONB (Postgres)

8. Constraints & Indexes (PK, FK, UNIQUE, CHECK, INDEX)

Constraints enforce data integrity rules. Indexes speed up queries — both are essential for production databases.

Command / SyntaxDescriptionExample
PRIMARY KEYUnique identifier for each row
▶ Show
-- Inline:
CREATE TABLE users (
  id INT PRIMARY KEY,
  name VARCHAR(100)
);

-- Named constraint:
CREATE TABLE users (
  id INT,
  name VARCHAR(100),
  CONSTRAINT pk_users PRIMARY KEY (id)
);

-- Composite PK:
CREATE TABLE order_items (
  order_id INT,
  product_id INT,
  PRIMARY KEY (order_id, product_id)
);
FOREIGN KEYEnforce referential integrity between tables
▶ Show
CREATE TABLE orders (
  id SERIAL PRIMARY KEY,
  customer_id INT,
  CONSTRAINT fk_customer
    FOREIGN KEY (customer_id)
    REFERENCES customers(id)
    ON DELETE CASCADE
    ON UPDATE SET NULL
);

-- Add FK to existing table:
ALTER TABLE orders
ADD CONSTRAINT fk_customer
  FOREIGN KEY (customer_id)
  REFERENCES customers(id);
UNIQUEEnsure no duplicate values in column(s)
▶ Show
CREATE TABLE users (
  id SERIAL PRIMARY KEY,
  email VARCHAR(255) UNIQUE,
  -- Composite unique:
  UNIQUE (first_name, last_name)
);

-- Add to existing table:
ALTER TABLE users
ADD CONSTRAINT uq_email UNIQUE (email);
CHECKValidate data against a condition
▶ Show
CREATE TABLE products (
  id SERIAL PRIMARY KEY,
  price DECIMAL(10,2) CHECK (price >= 0),
  quantity INT CHECK (quantity >= 0),
  CONSTRAINT chk_dates
    CHECK (end_date > start_date)
);

-- Add check to existing table:
ALTER TABLE employees
ADD CONSTRAINT chk_salary
  CHECK (salary > 0);
CREATE INDEXSpeed up queries on frequently searched columns
▶ Show
-- Single column index:
CREATE INDEX idx_emp_dept
ON employees(department);

-- Composite index:
CREATE INDEX idx_emp_dept_sal
ON employees(department, salary);

-- Unique index:
CREATE UNIQUE INDEX idx_users_email
ON users(email);

-- Partial index (PostgreSQL):
CREATE INDEX idx_active_emp
ON employees(name)
WHERE active = true;

-- Drop index:
DROP INDEX idx_emp_dept;

9. Views & Procedures (CREATE VIEW, Stored Procedures, Functions)

Views simplify complex queries into reusable abstractions. Procedures encapsulate business logic in the database.

Command / SyntaxDescriptionExample
CREATE VIEWVirtual table based on a query
▶ Show
CREATE VIEW active_employees AS
SELECT id, name, department, salary
FROM employees
WHERE active = true;

-- Use like a table:
SELECT * FROM active_employees
WHERE department = 'Engineering';

-- Replace existing view:
CREATE OR REPLACE VIEW active_employees AS
SELECT id, name, department, salary, hire_date
FROM employees
WHERE active = true;

-- Drop view:
DROP VIEW IF EXISTS active_employees;
Materialized ViewCached view — stores results physically
▶ Show
-- PostgreSQL:
CREATE MATERIALIZED VIEW monthly_sales AS
SELECT date_trunc('month', order_date) AS month,
       SUM(amount) AS total
FROM orders
GROUP BY 1;

-- Refresh data:
REFRESH MATERIALIZED VIEW monthly_sales;

-- Concurrent refresh (no lock):
REFRESH MATERIALIZED VIEW CONCURRENTLY
  monthly_sales;
Stored ProcedureReusable block of SQL logic (no return value)
▶ Show
-- PostgreSQL:
CREATE OR REPLACE PROCEDURE give_raise(
  emp_id INT,
  pct DECIMAL
)
LANGUAGE plpgsql AS $$
BEGIN
  UPDATE employees
  SET salary = salary * (1 + pct/100)
  WHERE id = emp_id;
  COMMIT;
END;
$$;

-- Call it:
CALL give_raise(42, 10);
FunctionReturns a value; can be used in queries
▶ Show
-- PostgreSQL:
CREATE OR REPLACE FUNCTION get_dept_avg(
  dept_name VARCHAR
) RETURNS DECIMAL AS $$
BEGIN
  RETURN (
    SELECT AVG(salary) FROM employees
    WHERE department = dept_name
  );
END;
$$ LANGUAGE plpgsql;

-- Use in query:
SELECT name, salary,
  get_dept_avg(department) AS dept_avg
FROM employees;

-- Drop function:
DROP FUNCTION IF EXISTS get_dept_avg;

10. Window Functions (ROW_NUMBER, RANK, DENSE_RANK, LAG, LEAD, NTILE)

Window functions perform calculations across related rows without collapsing them — rankings, running totals, row comparisons.

Command / SyntaxDescriptionExample
ROW_NUMBER()Sequential number for each row in partition
▶ Show
SELECT name, department, salary,
  ROW_NUMBER() OVER (
    PARTITION BY department
    ORDER BY salary DESC
  ) AS rn
FROM employees;

-- Top earner per department:
SELECT * FROM (
  SELECT *, ROW_NUMBER() OVER (
    PARTITION BY department
    ORDER BY salary DESC
  ) AS rn
  FROM employees
) t WHERE rn = 1;
RANK() / DENSE_RANK()Rank with gaps / without gaps for ties
▶ Show
SELECT name, salary,
  RANK() OVER (ORDER BY salary DESC) AS rnk,
  DENSE_RANK() OVER (ORDER BY salary DESC) AS drnk
FROM employees;
-- salary: 100k, 90k, 90k, 80k
-- RANK:      1,    2,    2,    4  (gap)
-- DENSE_RANK: 1,   2,    2,    3  (no gap)
LAG() / LEAD()Access previous/next row values
▶ Show
SELECT order_date, amount,
  LAG(amount, 1) OVER (ORDER BY order_date)
    AS prev_amount,
  LEAD(amount, 1) OVER (ORDER BY order_date)
    AS next_amount,
  amount - LAG(amount) OVER (ORDER BY order_date)
    AS change
FROM orders;

-- LAG(col, n, default) — n rows back
NTILE()Divide rows into N equal buckets
▶ Show
SELECT name, salary,
  NTILE(4) OVER (ORDER BY salary DESC)
    AS quartile
FROM employees;
-- Assigns 1-4 (top 25% = 1, etc.)
SUM/AVG OVERRunning totals and moving averages
▶ Show
-- Running total:
SELECT order_date, amount,
  SUM(amount) OVER (
    ORDER BY order_date
    ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
  ) AS running_total
FROM orders;

-- 3-day moving average:
SELECT order_date, amount,
  AVG(amount) OVER (
    ORDER BY order_date
    ROWS BETWEEN 2 PRECEDING AND CURRENT ROW
  ) AS moving_avg_3
FROM orders;
FIRST_VALUE / LAST_VALUEGet first or last value in window frame
▶ Show
SELECT name, department, salary,
  FIRST_VALUE(name) OVER (
    PARTITION BY department
    ORDER BY salary DESC
  ) AS top_earner
FROM employees;

11. CTEs & Recursive Queries

CTEs (WITH clause) make complex queries readable. Recursive CTEs traverse hierarchies like org charts and categories.

Command / SyntaxDescriptionExample
WITH (CTE)Named temporary result set for readability
▶ Show
WITH dept_stats AS (
  SELECT department,
         AVG(salary) AS avg_sal,
         COUNT(*) AS cnt
  FROM employees
  GROUP BY department
)
SELECT * FROM dept_stats
WHERE avg_sal > 70000
ORDER BY avg_sal DESC;
Multiple CTEsChain multiple CTEs separated by commas
▶ Show
WITH
high_earners AS (
  SELECT * FROM employees
  WHERE salary > 100000
),
dept_counts AS (
  SELECT department, COUNT(*) AS cnt
  FROM high_earners
  GROUP BY department
)
SELECT * FROM dept_counts
ORDER BY cnt DESC;
RECURSIVE CTESelf-referencing query for hierarchical data
▶ Show
-- Org chart traversal:
WITH RECURSIVE org_tree AS (
  -- Base case: top-level (no manager)
  SELECT id, name, manager_id, 1 AS level
  FROM employees
  WHERE manager_id IS NULL

  UNION ALL

  -- Recursive case: find direct reports
  SELECT e.id, e.name, e.manager_id, t.level + 1
  FROM employees e
  JOIN org_tree t ON e.manager_id = t.id
)
SELECT * FROM org_tree
ORDER BY level, name;
Recursive — Number SeriesGenerate sequences without a table
▶ Show
-- Generate numbers 1-100:
WITH RECURSIVE nums AS (
  SELECT 1 AS n
  UNION ALL
  SELECT n + 1 FROM nums WHERE n < 100
)
SELECT n FROM nums;

-- Generate date series:
WITH RECURSIVE dates AS (
  SELECT '2024-01-01'::DATE AS d
  UNION ALL
  SELECT d + 1 FROM dates WHERE d < '2024-12-31'
)
SELECT d FROM dates;
CTE with DMLUse CTE with INSERT, UPDATE, DELETE
▶ Show
-- Delete and return deleted rows:
WITH deleted AS (
  DELETE FROM logs
  WHERE created_at < '2023-01-01'
  RETURNING *
)
INSERT INTO archive_logs
SELECT * FROM deleted;

12. Transactions (BEGIN, COMMIT, ROLLBACK, SAVEPOINT, Isolation)

Transactions ensure data consistency — either all changes succeed together or none apply (ACID properties).

Command / SyntaxDescriptionExample
BEGIN / COMMITStart and complete a transaction
▶ Show
BEGIN;  -- or START TRANSACTION

UPDATE accounts SET balance = balance - 500
WHERE id = 1;

UPDATE accounts SET balance = balance + 500
WHERE id = 2;

COMMIT;  -- both succeed or neither does
ROLLBACKUndo all changes since BEGIN
▶ Show
BEGIN;

DELETE FROM orders WHERE status = 'draft';
-- Oops, wrong condition!

ROLLBACK;  -- nothing deleted
SAVEPOINTCreate a named point to partially rollback to
▶ Show
BEGIN;

INSERT INTO orders (customer_id, amount)
VALUES (1, 100);

SAVEPOINT sp1;

INSERT INTO order_items (order_id, product_id)
VALUES (999, 5);  -- might fail

-- If error, rollback only to savepoint:
ROLLBACK TO SAVEPOINT sp1;

-- First insert still intact:
COMMIT;
Isolation LevelsControl visibility of concurrent changes
▶ Show
-- Set for current transaction:
SET TRANSACTION ISOLATION LEVEL
  READ COMMITTED;

-- Levels (low → high isolation):
-- READ UNCOMMITTED  — dirty reads possible
-- READ COMMITTED    — default in PostgreSQL
-- REPEATABLE READ   — default in MySQL
-- SERIALIZABLE      — strongest, slowest

-- PostgreSQL example:
BEGIN ISOLATION LEVEL SERIALIZABLE;
SELECT * FROM inventory
  WHERE product_id = 42;
UPDATE inventory SET qty = qty - 1
  WHERE product_id = 42;
COMMIT;
LOCKExplicitly lock tables or rows
▶ Show

Index StrategyWhen and how to create effective indexes
▶ Show
-- Index columns used in:
--   WHERE, JOIN ON, ORDER BY, GROUP BY

-- Composite index order matters:
CREATE INDEX idx_dept_sal
ON employees(department, salary);
-- Helps: WHERE dept = 'X' AND salary > N
-- Helps: WHERE dept = 'X' ORDER BY salary
-- No help: WHERE salary > N (2nd col only)

-- Covering index (includes all needed cols):
CREATE INDEX idx_cover
ON orders(customer_id)
INCLUDE (order_date, amount);

-- Check unused indexes (PostgreSQL):
SELECT indexrelname, idx_scan
FROM pg_stat_user_indexes
WHERE idx_scan = 0;
Query Optimization TipsCommon patterns to improve performance
▶ Show
-- ❌ Avoid SELECT * in production:
SELECT id, name, email FROM users;

-- ❌ Avoid functions on indexed columns:
-- Bad:  WHERE YEAR(created_at) = 2024
-- Good: WHERE created_at >= '2024-01-01'
--         AND created_at < '2025-01-01'

-- ❌ Avoid OR on different columns (use UNION):
SELECT * FROM emp WHERE dept_id = 1
UNION ALL
SELECT * FROM emp WHERE mgr_id = 5;

-- ✅ EXISTS vs IN for large subqueries:
SELECT * FROM customers c
WHERE EXISTS (
  SELECT 1 FROM orders o
  WHERE o.cust_id = c.id
);

-- ✅ Use LIMIT for pagination:
SELECT * FROM products
ORDER BY id LIMIT 20 OFFSET 40;

Advanced SQL techniques handle complex reporting, JSON data, temporary storage, and automated triggers.

Command / SyntaxDescriptionExample
PIVOT / CROSSTABTranspose rows into columns
▶ Show
-- PostgreSQL (crosstab extension):
SELECT * FROM crosstab(
  'SELECT department, quarter, revenue
   FROM sales ORDER BY 1,2'
) AS ct(dept TEXT, q1 INT, q2 INT, q3 INT, q4 INT);

-- Manual pivot with CASE:
SELECT department,
  SUM(CASE WHEN quarter=1 THEN revenue END) AS q1,
  SUM(CASE WHEN quarter=2 THEN revenue END) AS q2,
  SUM(CASE WHEN quarter=3 THEN revenue END) AS q3,
  SUM(CASE WHEN quarter=4 THEN revenue END) AS q4
FROM sales
GROUP BY department;

-- SQL Server PIVOT:
SELECT * FROM sales
PIVOT (
  SUM(revenue) FOR quarter IN ([1],[2],[3],[4])
) AS pvt;

TriggersAuto-execute code on INSERT/UPDATE/DELETE
▶ Show
-- PostgreSQL trigger function:
CREATE OR REPLACE FUNCTION update_modified()
RETURNS TRIGGER AS $$
BEGIN
  NEW.updated_at = NOW();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Attach trigger:
CREATE TRIGGER trg_update_modified
BEFORE UPDATE ON employees
FOR EACH ROW
EXECUTE FUNCTION update_modified();

-- Audit trigger:
CREATE TRIGGER trg_audit
AFTER DELETE ON orders
FOR EACH ROW
EXECUTE FUNCTION log_deletion();

-- Drop trigger:
DROP TRIGGER trg_update_modified
ON employees;
